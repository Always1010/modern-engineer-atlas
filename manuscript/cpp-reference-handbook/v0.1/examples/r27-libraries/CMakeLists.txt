cmake_minimum_required(VERSION 3.16)
project(r27_libraries LANGUAGES CXX)
if(NOT WIN32 AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "This example supports Windows and Linux only")
endif()
if(WIN32 AND NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The Windows example requires 64-bit targets; Win32 export decoration is not covered")
endif()

add_library(metric_static STATIC metric.cpp)
target_compile_definitions(metric_static PUBLIC METRIC_STATIC)
add_library(metric_shared SHARED metric.cpp)
target_compile_definitions(metric_shared PRIVATE METRIC_BUILD)
add_library(metric_plugin MODULE metric.cpp)
target_compile_definitions(metric_plugin PRIVATE METRIC_BUILD)
foreach(target metric_static metric_shared metric_plugin)
    target_include_directories(${target} PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
    target_compile_features(${target} PUBLIC cxx_std_17)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF
        CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
endforeach()

add_executable(static_app linked_main.cpp)
target_link_libraries(static_app PRIVATE metric_static)
add_executable(shared_app linked_main.cpp)
target_link_libraries(shared_app PRIVATE metric_shared)
add_executable(plugin_host plugin_main.cpp)
target_compile_definitions(plugin_host PRIVATE METRIC_EXPLICIT)
target_include_directories(plugin_host PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(plugin_host PRIVATE cxx_std_17)
if(WIN32)
    target_link_libraries(plugin_host PRIVATE shell32)
else()
    target_link_libraries(plugin_host PRIVATE ${CMAKE_DL_LIBS})
endif()
foreach(target static_app shared_app plugin_host)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
endforeach()
# DLL and applications share the default build output directory on Windows.
# CMake's build RPATH finds metric_shared on Linux; this is not an install policy.
enable_testing()
add_test(NAME r27_static COMMAND static_app)
add_test(NAME r27_shared COMMAND shared_app)
add_test(NAME r27_plugin COMMAND plugin_host $<TARGET_FILE:metric_plugin>)
